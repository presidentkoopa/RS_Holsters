// SafeCommand rather than Command for the destructive/modal ones: edit
// mode and reset both change state you would not want triggered by a
// misread controller press while scrolling a menu in VR.
OptionMenu "RS_HolsterPlacement"
{
	Title "Placement & Profiles"
	StaticText " "

	StaticText "Placement", "Gold"
	Option "How many holsters",         "rs_holster_active_count", "RS_HolsterActiveCount"
	SafeCommand "Edit Mode -- grab spheres and move them", "netevent rs-holster-edit"
	SafeCommand "Print tuned table to console",            "netevent rs-holster-table"
	SafeCommand "Reset offsets to defaults",               "netevent rs-holster-reset"
	SafeCommand "Recalibrate standing height",             "netevent rs-holster-recalibrate"
	StaticText " "

	// A profile is the CURRENT edit-table -- tune it with Edit Mode above,
	// then save it under whichever posture you tuned it for. Loading takes
	// effect immediately; nothing here needs a rebuild or a relaunch.
	StaticText "Profiles", "Gold"
	SafeCommand "Save current as STANDING", "netevent rs-holster-save-standing"
	SafeCommand "Load STANDING",            "netevent rs-holster-load-standing"
	SafeCommand "Save current as SEATED",   "netevent rs-holster-save-seated"
	SafeCommand "Load SEATED",              "netevent rs-holster-load-seated"
	StaticText " "
	SafeCommand "SWITCH seated/standing",   "netevent rs-holster-switch-profile"
}

OptionMenu "RS_HolsterTiming"
{
	Title "Timing & Grip"
	StaticText " "

	StaticText "Timing", "Gold"
	// Uses GZDoom's own instant-switch mechanism, scoped to just the swap a
	// holster triggers -- number-key and wheel switching keep their normal
	// animated speed either way. Off restores the real ~16-tic lower/raise
	// and a longer debounce between store/draw presses.
	Option "Instant store/draw", "rs_holster_instant_switch", "OnOff"
	StaticText " "

	StaticText "Grip", "Gold"
	// When on, a hand inside a holster spends its grip on the holster and
	// nothing else -- stabilize and the shift layer both stand down for
	// that hand. Off leaves grip alone entirely and the bound holster keys
	// are the only way in. These three are ENGINE cvars (DoomXR), not this
	// pk3's own -- referencing them here does not require this pk3 to own
	// or declare them.
	Option "Grip works holsters",       "vr_holster_use_grip", "OnOff"
	Option "Two handed weapons",        "vr_two_handed_weapons", "OnOff"
	Slider "Stabilize distance (in)",   "vr_stabilize_distance_inches", 1.0, 24.0, 1.0, 0
}
