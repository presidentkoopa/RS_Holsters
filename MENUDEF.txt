// SafeCommand rather than Command for the destructive/modal ones: edit
// mode and reset both change state you would not want triggered by a
// misread controller press while scrolling a menu in VR.

{
	Title "Placement & Profiles"
	StaticText " "

	StaticText "Placement", "Gold"
	Option "How many holsters",         "rs_hardpoint_active_count", "RS_HardPointActiveCount"
	SafeCommand "Edit Mode -- grab spheres and move them", "netevent rs-hardpoint-edit"
	SafeCommand "Print tuned table to console",            "netevent rs-hardpoint-table"
	SafeCommand "Reset offsets to defaults",               "netevent rs-hardpoint-reset"
	SafeCommand "Recalibrate standing height",             "netevent rs-hardpoint-recalibrate"
	StaticText " "

	// A profile is the CURRENT edit-table -- tune it with Edit Mode above,
	// then save it under whichever posture you tuned it for. Loading takes
	// effect immediately; nothing here needs a rebuild or a relaunch.
	StaticText "Profiles", "Gold"
	SafeCommand "Save current as STANDING", "netevent rs-hardpoint-save-standing"
	SafeCommand "Load STANDING",            "netevent rs-hardpoint-load-standing"
	SafeCommand "Save current as SEATED",   "netevent rs-hardpoint-save-seated"
	SafeCommand "Load SEATED",              "netevent rs-hardpoint-load-seated"
	StaticText " "
	SafeCommand "SWITCH seated/standing",   "netevent rs-hardpoint-switch-profile"
}

OptionMenu "RS_HardPointTiming"
{
	Title "Timing & Grip"
	StaticText " "

	StaticText "Timing", "Gold"
	// Uses GZDoom's own instant-switch mechanism, scoped to just the swap a
	// holster triggers -- number-key and wheel switching keep their normal
	// animated speed either way. Off restores the real ~16-tic lower/raise
	// and a longer debounce between store/draw presses.
	Option "Instant store/draw", "rs_hardpoint_instant_switch", "OnOff"
	StaticText " "

	StaticText "Grip", "Gold"
	// When on, a hand inside a holster spends its grip on the holster and
	// nothing else -- stabilize and the shift layer both stand down for
	// that hand. Off leaves grip alone entirely and the bound holster keys
	// are the only way in. These three are ENGINE cvars (DoomXR), not this
	// pk3's own -- referencing them here does not require this pk3 to own
	// or declare them.
	Option "Grip works holsters",       "vr_holster_use_grip", "OnOff"
	Option "Two handed weapons",        "vr_two_handed_weapons", "OnOff"
	Slider "Stabilize distance (in)",   "vr_stabilize_distance_inches", 1.0, 24.0, 1.0, 0
}

// Arm Rig: the off-hand-anchored forearm/wrist hardpoints (GetHolster
// indices 8-13). Separate page from Body Holsters because it is a
// separate on/off tier (rs_hardpoint_arm_active_count, defaults OFF) -- new,
// unproven-in-headset anchor math next to the already-tuned 8.
OptionMenu "RS_ArmRigOptions"
{
	Title "Arm Rig (Forearm & Wrist)"
	StaticText " "

	StaticText "Placement", "Gold"
	Option "How many hardpoints", "rs_hardpoint_arm_active_count", "RS_HardPointArmActiveCount"
	StaticText " "

	StaticText "Appearance", "Gold"
	Slider "Hardpoint item size", "rs_hardpoint_prop_scale_arm", 0.02, 0.5, 0.01, 2
	StaticText " "

	// Generic over HOLSTER_COUNT already -- these reach indices 8-13 for
	// free, same netevents Body Holsters -> Placement & Profiles uses.
	// Duplicated here for discoverability, not because they need separate
	// logic.
	SafeCommand "Edit Mode -- grab spheres and move them", "netevent rs-hardpoint-edit"
	SafeCommand "Print tuned table to console",            "netevent rs-hardpoint-table"
	StaticText " "

	// NOT duplicating "Reset offsets" or the save/load-profile buttons
	// here on purpose: rs-hardpoint-reset reseeds ALL 14 holsters in one
	// shot, not just these 6 -- a Reset button on this page would read as
	// scoped to the arm rig and silently wipe an already-tuned body table
	// too. Said explicitly rather than letting that surprise someone.
	StaticText "Profiles and offset RESET live on the Body Holsters", "Gold"
	StaticText "page -- Reset there clears all 14 holsters, not just these.", "Gold"
}
